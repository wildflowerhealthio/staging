Wildflower, Read in Order

Session Shortfall

Covers authorization-server-rust/tests/integration/session_shortfall.rs

A browser paired with read-only access may run an app that asks for more. The code it earns would carry only what its session holds, so the browser is told first, on the limited page, before anyone else is bothered. /oauth/authorize parks the request awaiting that page and sends the browser to /oauth/authorize/{id}/limited. The page reads what the session lacks (session_shortfall), then goes on (…/continue) or cancels (…/cancel).

Until the browser goes on, the Owner isn’t asked and nothing decides the request. Going on takes the branch the start would have taken: the fast path under a standing grant, the consent card, or the wait page. Only the browser that made the request can read, continue or cancel it. Once a browser session has gone on for an app, its later requests for that app skip the page. The request’s two limited-page transitions are CodeRequest::acknowledge_session_shortfall and CodeRequest::deny_on_limited_page.

the_fast_path_stops_at_the_limited_page_then_issues_the_narrowed_code

A standing grant covers the request, but the browser’s session falls short of it: the request parks on the limited page with nothing issued and nothing raised, the page reads the app and what the session lacks, and going on issues a code narrowed to the session.

#![allow(unused)]
fn main() {
async fn the_fast_path_stops_at_the_limited_page_then_issues_the_narrowed_code() {
    let (g, owner_token, db) = spin_up();
    seed_app(&db);
    approve_a_standing_grant(&g, &owner_token, &db).await;
    let (_, session_token) = limited_browser(&db);

    let request_id =
        limited_request_id(&get_authorize(&g.router, &session_token, &app_query()).await);
    assert_eq!(owner_prompt(&db), None, "the Owner isn't asked");
    assert_eq!(
        store_handle(&db)
            .read(|tx| tx.authorization_code_by_request_id(&request_id))
            .expect("read"),
        None,
        "no code yet"
    );

    let res = get_session_shortfall(&g.router, &session_token, &request_id).await;
    assert_eq!(res.status(), StatusCode::OK);
    assert_cache_suppressed(&res, "the limited page's read");
    assert_eq!(
        body_json(res.into_body()).await,
        serde_json::json!({
            "clientName": "Integration Test Client",
            "missingScopes": ["patient/Observation.rs"],
        })
    );

    let callback = continue_past_session_shortfall(&g.router, &session_token, &request_id).await;
    assert!(
        callback.starts_with("https://app.example/cb?code="),
        "{callback}"
    );
    let issued = redeem(&g, &code_of(&callback)).await;
    assert_eq!(issued["scope"], "patient/Patient.rs offline_access");
    assert_eq!(owner_prompt(&db), None);

    let res = post_session_shortfall(&g.router, &session_token, &request_id, "continue").await;
    assert_eq!(
        res.status(),
        StatusCode::NOT_FOUND,
        "a request goes on once"
    );
}
}

a_session_that_went_on_for_an_app_skips_the_limited_page_for_it

Once a browser session has gone on past the limited page for an app, its next request for that app goes straight on (here the fast path); another session, and the same session asking for another app, still stop at it.

#![allow(unused)]
fn main() {
async fn a_session_that_went_on_for_an_app_skips_the_limited_page_for_it() {
    let (g, owner_token, db) = spin_up();
    seed_app(&db);
    seed_client_with_redirect(
        &db,
        "other-app",
        "https://app.example/cb",
        &[
            "patient/Patient.rs",
            "patient/Observation.rs",
            "offline_access",
        ],
    );
    approve_a_standing_grant(&g, &owner_token, &db).await;
    let (_, session_token) = limited_browser(&db);
    let request_id =
        limited_request_id(&get_authorize(&g.router, &session_token, &app_query()).await);
    continue_past_session_shortfall(&g.router, &session_token, &request_id).await;

    let res = get_authorize(&g.router, &session_token, &app_query()).await;
    assert_eq!(res.status(), StatusCode::FOUND);
    let callback = location_of(&res);
    assert!(
        callback.starts_with("https://app.example/cb?code="),
        "{callback}"
    );
    let issued = redeem(&g, &code_of(&callback)).await;
    assert_eq!(issued["scope"], "patient/Patient.rs offline_access");

    let (_, other_session_token) = limited_browser(&db);
    limited_request_id(&get_authorize(&g.router, &other_session_token, &app_query()).await);
    let other_app_query = authorize_query(
        "other-app",
        "patient%2FPatient.rs%20patient%2FObservation.rs%20offline_access",
    );
    limited_request_id(&get_authorize(&g.router, &session_token, &other_app_query).await);
}
}

the_owner_neither_sees_nor_decides_a_request_waiting_on_its_limited_page

A request waiting on its limited page isn’t the Owner’s to see or decide: the Owner prompt skips it, and the consent read, approve, deny and ask-owner all 404. Once the browser goes on, the Owner sees and approves it.

#![allow(unused)]
fn main() {
async fn the_owner_neither_sees_nor_decides_a_request_waiting_on_its_limited_page() {
    let (g, owner_token, db) = spin_up();
    seed_app(&db);
    let (_, session_token) = limited_browser(&db);
    let request_id =
        limited_request_id(&get_authorize(&g.router, &session_token, &app_query()).await);

    assert_eq!(owner_prompt(&db), None);
    let owner_request = |method: &str, action: &str| {
        let uri = format!("/authorization/oauth-consents/{request_id}{action}");
        build_request(
            Request::builder()
                .method(method)
                .uri(uri)
                .header("authorization", format!("Bearer {owner_token}"))
                .header("content-type", "application/json"),
            Body::from(if action == "/approve" {
                APPROVE_APP
            } else {
                ""
            }),
        )
    };
    for (method, action) in [
        ("GET", ""),
        ("POST", "/approve"),
        ("POST", "/deny"),
        ("POST", "/ask-owner"),
    ] {
        let res = g
            .router
            .clone()
            .oneshot(owner_request(method, action))
            .await
            .expect("oneshot");
        assert_eq!(res.status(), StatusCode::NOT_FOUND, "{method} {action}");
    }

    assert_eq!(
        continue_past_session_shortfall(&g.router, &session_token, &request_id).await,
        format!("/oauth/authorize/{request_id}/wait")
    );
    assert!(matches!(
        owner_prompt(&db),
        Some(OwnerPrompt::AuthorizationCode { ref id, .. }) if *id == request_id
    ));
    let res = approve_oauth_consent(&g, &owner_token, &request_id, APPROVE_APP.to_owned()).await;
    assert_eq!(res.status(), StatusCode::OK);
}
}

another_browser_can_neither_read_nor_continue_nor_cancel

Another browser’s session, and a browser without one, get the 404 an unknown id gets on the read, Continue and Cancel, and the request still waits for its own browser.

#![allow(unused)]
fn main() {
async fn another_browser_can_neither_read_nor_continue_nor_cancel() {
    let (g, _owner_token, db) = spin_up();
    seed_app(&db);
    let (_, session_token) = limited_browser(&db);
    let request_id =
        limited_request_id(&get_authorize(&g.router, &session_token, &app_query()).await);
    let (_, other_browser_token) = limited_browser(&db);
    let same_device_session_token = signed_in_same_device_session_token(&g).await;

    for token in [&other_browser_token, &same_device_session_token] {
        let res = get_session_shortfall(&g.router, token, &request_id).await;
        assert_eq!(res.status(), StatusCode::NOT_FOUND);
        assert_eq!(
            body_json(res.into_body()).await["error"],
            "AuthorizationRequestNotFound"
        );
        for action in ["continue", "cancel"] {
            let res = post_session_shortfall(&g.router, token, &request_id, action).await;
            assert_eq!(res.status(), StatusCode::NOT_FOUND, "{action}");
        }
    }
    for (method, path) in [
        (
            "GET",
            format!("/oauth/authorize/{request_id}/session-shortfall"),
        ),
        ("POST", format!("/oauth/authorize/{request_id}/continue")),
        ("POST", format!("/oauth/authorize/{request_id}/cancel")),
    ] {
        let res = g
            .router
            .clone()
            .oneshot(build_request(
                Request::builder().method(method).uri(&path),
                Body::empty(),
            ))
            .await
            .expect("oneshot");
        assert_eq!(res.status(), StatusCode::NOT_FOUND, "{method} {path}");
    }

    let res = get_session_shortfall(&g.router, &session_token, &request_id).await;
    assert_eq!(
        res.status(),
        StatusCode::OK,
        "it still waits for its browser"
    );
}
}

an_expired_request_on_the_limited_page_goes_no_further

A request whose five minutes ran out on the limited page can’t be read, continued or cancelled, and the status poll reports it expired, as for any other request.

#![allow(unused)]
fn main() {
async fn an_expired_request_on_the_limited_page_goes_no_further() {
    let (g, _owner_token, db) = spin_up();
    seed_app(&db);
    let (_, session_token) = limited_browser(&db);
    let request_id =
        limited_request_id(&get_authorize(&g.router, &session_token, &app_query()).await);
    let parked = store_handle(&db)
        .read(|tx| tx.code_request_by_id(&request_id))
        .expect("read")
        .expect("parked");
    let lapsed_at = Utc::now() - Duration::minutes(5) - Duration::seconds(1);
    store_handle(&db)
        .write(|tx| {
            tx.insert_code_request(&CodeRequest {
                id: "lapsed".to_owned(),
                requested_at: lapsed_at,
                expires_at: lapsed_at + Duration::minutes(5),
                ..parked
            })
        })
        .expect("plant a lapsed copy");

    for res in [
        get_session_shortfall(&g.router, &session_token, "lapsed").await,
        post_session_shortfall(&g.router, &session_token, "lapsed", "continue").await,
        post_session_shortfall(&g.router, &session_token, "lapsed", "cancel").await,
    ] {
        assert_eq!(res.status(), StatusCode::NOT_FOUND);
    }
    let res = get_authorization_status(&g.router, &session_token, "lapsed").await;
    assert_eq!(body_json(res.into_body()).await["status"], "error");
}
}
