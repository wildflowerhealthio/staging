Wildflower, Read in Order

OAuth Endpoints

Covers authorization-server-rust/tests/integration/oauth_endpoints.rs

These tests hold the server’s three OAuth doors to their contract with apps: /oauth/authorize, where a browser starts an app’s request, /oauth/device_authorization, where a device starts one, and /oauth/token, where an app collects what it was granted. They are about what each door accepts, refuses and reports, more than about what happens once the Owner decides; Grant Flows follows the flows end to end.

Most of the file is /oauth/authorize, and most of that is one rule from RFC 6749: an error may be sent back to an app’s redirect only once that redirect is trusted. A redirect is trusted when a client the server already knows has it registered (Client::allows_redirect_uri). Until then, a failure renders a local error page (AuthorizeRefusal), because redirecting to an address nobody vouched for would be an open redirect. Clients without a server-defined registration are trusted on first use: an unknown client, a new redirect or a new scope is carried to the Owner’s prompt as a presented client change instead of refused.

The tests run in this order: what /oauth/authorize checks and how it fails, a request’s scopes against its client’s registration, telling the Owner about a parked request, SMART App Launch (launch context), the device door, and the token door.

authorize_unknown_client_parks_a_pending_request

Trust on first use: a client_id this authorization server has never seen isn’t rejected at the door. Its request is parked like any other and the browser is sent to wait for the Owner, and nothing is written to the clients table, so a request nobody approves leaves no trace of the app.

#![allow(unused)]
fn main() {
async fn authorize_unknown_client_parks_a_pending_request() {
    let (g, _owner_token, db) = spin_up();
    let res = get_authorize(
        &g.router,
        &db.paired_browser_token,
        &authorize_query("ghost", "read"),
    )
    .await;
    let request_id = parked_request_id(&res);

    let res = get_authorization_status(&g.router, &db.paired_browser_token, &request_id).await;
    assert_eq!(res.status(), StatusCode::OK);
    assert_eq!(body_json(res.into_body()).await["status"], "pending");
    assert!(
        store_handle(&db)
            .read(|tx| tx.client_by_id("ghost"))
            .expect("client query")
            .is_none(),
        "/authorize must not register an unknown client",
    );
}
}

authorize_unknown_client_renders_later_failures_locally

The redirect of an unknown client is untrusted, so a later validation failure may NOT be 302’d to it (RFC 6749 §4.1.2.1’s open-redirect rule): a malformed PKCE challenge and a non-code response_type both render the local HTML page instead.

#![allow(unused)]
fn main() {
async fn authorize_unknown_client_renders_later_failures_locally() {
    let (g, _owner_token, db) = spin_up();
    let challenge = compute_code_challenge(CODE_VERIFIER);
    for (query, expected) in [
        (
            "response_type=code&code_challenge_method=S256&client_id=ghost&scope=read&\
             code_challenge=abc&redirect_uri=https%3A%2F%2Fapp.example%2Fcb&state=xyz"

the_pending_consent_watcher_sees_an_older_request_decided_under_a_newer_one

The host’s PendingConsentWatcher sees a request settle even when it isn’t the newest: denying an older request while a newer one waits leaves the Owner prompt unchanged, so nothing reaches the host through that.

#![allow(unused)]
fn main() {
async fn the_pending_consent_watcher_sees_an_older_request_decided_under_a_newer_one() {
    let (g, owner_token, db) = spin_up();
    seed_client_with_redirect(&db, "test-app", "https://app.example/cb", &["read"]);
    let older_id = parked_request_id(
        &get_authorize(
            &g.router,
            &db.paired_browser_token,
            &authorize_query("test-app", "read"),
        )
        .await,
    );
    let newer_id = parked_request_id(
        &get_authorize(
            &g.router,
            &db.paired_browser_token,
            &authorize_query("test-app", "read"),
        )
        .await,
    );
    let watcher = g.pending_consent_watcher.clone();
    let older_key = ConsentKey::AuthorizationCode {
        id: older_id.clone(),
    };
    let older_settled = tokio::spawn(async move { watcher.until_settled(older_key).await });
    tokio::time::sleep(std::time::Duration::from_millis(50)).await;
    assert!(
        !older_settled.is_finished(),
        "the older request still waits"
    );

    let res = g
        .router
        .clone()
        .oneshot(build_request(
            Request::post(format!("/authorization/oauth-consents/{older_id}/deny"))
                .header("host", "127.0.0.1")
                .header("authorization", format!("Bearer {owner_token}"))
                .header("content-type", "application/json"),
            Body::from(r#"{"approvedScopes":["read"],"acknowledgedRegistrationChange":true}"#),
        ))
        .await
        .expect("oneshot");
    assert_eq!(res.status(), StatusCode::OK);

    tokio::time::timeout(std::time::Duration::from_secs(5), older_settled)
        .await
        .expect("the watcher sees the older request settle")
        .expect("the watcher's task")
        .expect("the watcher's reads");
    assert_eq!(
        owner_prompt(&db),
        Some(OwnerPrompt::AuthorizationCode {
            id: newer_id,
            client_name: "Integration Test Client".to_owned(),
            decidable_in_requesting_browser: false,
        }),
        "the newer request is still the newest",

#![allow(unused)]
fn main() {
async fn device_authorization_happy_path() {
    let (g, _owner_token, db) = spin_up();
    seed_test_client(&db);
    let body = "client_id=integration-test-app&scope=system%2F*.cruds";
    let req = build_request(
        Request::post("/oauth/device_authorization")
            .header("content-type", "application/x-www-form-urlencoded"),
        Body::from(body),
    );
    let res = g.router.oneshot(req).await.expect("oneshot");
    assert_eq!(res.status(), StatusCode::OK);
    let body = body_json(res.into_body()).await;
    // device_code/user_code are random, so thread them through; the rest of
    // the RFC 8628 §3.2 body (verification_uri, verification_uri_complete,
    // expires_in, interval) is pinned by the whole-value comparison.
    let device_code = body["device_code"].clone();
    let user_code = body["user_code"].as_str().expect("user_code").to_string();
    assert_eq!(

            "user_code": user_code,
            // This server's own device page, on its origin.
            "verification_uri": "https://ruth.relay.example/oauth/device",
            "verification_uri_complete": format!(
                "https://ruth.relay.example/oauth/device?user_code={user_code}"
            ),
            "expires_in": 300,
            "interval": 5,
        })
    );
    // The user_code is also a well-formed, human-typeable pairing code — a
    // relationship the opaque whole-value comparison above can't express.
    assert!(
        wildflowerhealthio_authorization_server::crypto_util::oauth_user_code::is_valid_oauth_user_code(
            &user_code
        )
