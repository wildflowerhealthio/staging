Revocation

Covers authorization-server-rust/tests/integration/revocation.rs

An access token is a signed JWT that stays valid until it expires. Revocation is how the server stops one early. It keeps two records, in a revocation store shared with the FHIR server:

  • The jti denylist ends one token: the row lives until the token would have expired anyway.
  • The subject epoch ends a cohort: every token issued to a subject (its sub, the client_id) before the epoch is revoked.

Every /authorization bearer gate checks both, and a revoked token is 401, the same as no token. Revocation is written in three places: the Owner’s POST /authorization/revocations, logout, and revoking a grant. Token Revocation covers an app revoking its own tokens.

This file’s own helpers are on the Test Fixtures page, with the shared ones.

gate_rejects_a_revoked_jti

A live token whose jti is denylisted is rejected from then on.

#![allow(unused)]
fn main() {
async fn gate_rejects_a_revoked_jti() {
    let (g, owner_token, db) = spin_up();
    // Live first.
    let res = g
        .router
        .clone()
        .oneshot(owner_grants_probe(&owner_token))
        .await
        .expect("oneshot");
    assert_eq!(res.status(), StatusCode::OK);
    // Denylist this token's jti; the same token is now rejected.
    let jti = jti_of(&owner_token);
    revocation_store_handle(&db)
        .revoke_jti(&jti, Utc::now() + Duration::hours(1), "test")
        .expect("revoke");
    let res = g
        .router
        .clone()
        .oneshot(owner_grants_probe(&owner_token))
        .await
        .expect("oneshot");
    assert_eq!(
        res.status(),
        StatusCode::UNAUTHORIZED,
        "a revoked jti must be rejected by the gate"
    );
}
}

gate_rejects_after_subject_epoch_bump

Bumping a subject’s revocation epoch rejects every token issued to that client before the epoch. The epoch is set a second ahead, because a token’s iat is whole seconds.

#![allow(unused)]
fn main() {
async fn gate_rejects_after_subject_epoch_bump() {
    let (g, owner_token, db) = spin_up();
    // Bulk-revoke the Owner token's client's cohort. JWT `iat` is second-precision, so
    // advance the epoch one second past the mint second to unambiguously cover a
    // token minted moments ago in this test.
    revocation_store_handle(&db)
        .bump_subject_epoch(OWNER_TOKEN_CLIENT_ID, Utc::now() + Duration::seconds(1))
        .expect("bump");
    let res = g
        .router
        .clone()
        .oneshot(owner_grants_probe(&owner_token))
        .await
        .expect("oneshot");
    assert_eq!(
        res.status(),
        StatusCode::UNAUTHORIZED,
        "an epoch bump must revoke pre-epoch tokens"
    );
}
}

logout_revokes_the_presented_jti

Logging out denylists the presented token’s jti, so the same token no longer authenticates.

#![allow(unused)]
fn main() {
async fn logout_revokes_the_presented_jti() {
    let (g, owner_token, db) = spin_up();
    let jti = jti_of(&owner_token);
    // Log out, presenting the token as a bearer.
    let res = g
        .router
        .clone()
        .oneshot(build_request(
            Request::post("/authorization/logout")
                .header("host", "127.0.0.1")
                .header("authorization", format!("Bearer {owner_token}")),
            Body::empty(),
        ))
        .await
        .expect("oneshot");
    assert_eq!(res.status(), StatusCode::SEE_OTHER);
    // The presented token's jti is now denylisted...
    assert!(
        revocation_store_handle(&db)
            .is_revoked_by_jti(&jti)
            .expect("query"),
        "logout must denylist the presented token's jti"
    );
    // ...and the same token no longer authenticates.
    let res = g
        .router
        .clone()
        .oneshot(owner_grants_probe(&owner_token))
        .await
        .expect("oneshot");
    assert_eq!(res.status(), StatusCode::UNAUTHORIZED);
}
}

revocations_endpoint_rejects_invalid_bodies

A revocation body is one mode or the other: a jti without its expiresAt, both modes, neither, or an expiresAt already past is 400.

#![allow(unused)]
fn main() {
async fn revocations_endpoint_rejects_invalid_bodies() {
    let (g, owner_token, _db) = spin_up();
    let bad_bodies = [
        // jti without the required expiresAt
        serde_json::json!({ "jti": "abc" }),
        // both modes at once
        serde_json::json!({ "jti": "abc", "expiresAt": Utc::now().to_rfc3339(), "subject": "c" }),
        // neither mode
        serde_json::json!({}),
        // jti with an expiresAt already in the past (stale/mistyped)
        serde_json::json!({
            "jti": "abc",
            "expiresAt": (Utc::now() - Duration::hours(1)).to_rfc3339(),
        }),
    ];
    for body in bad_bodies {
        let res = g
            .router
            .clone()
            .oneshot(owner_revocation_request(&owner_token, body.clone()))
            .await
            .expect("oneshot");
        assert_eq!(
            res.status(),
            StatusCode::BAD_REQUEST,
            "invalid revocation body must 400: {body}"
        );
    }
}
}

revoking_a_grant_bumps_the_client_revocation_epoch

Revoking an authorization-code grant bumps its client’s revocation epoch, so the client’s live access tokens end with the standing consent.

#![allow(unused)]
fn main() {
async fn revoking_a_grant_bumps_the_client_revocation_epoch() {
    let (g, owner_token, db) = spin_up();
    // Seed a client + a grant for it.
    seed_client_with_redirect(&db, "granted-client", "https://app.example/cb", &["read"]);
    let store = store_handle(&db);
    store
        .write(|tx| {
            tx.create_authorization_code_grant(
                &wildflowerhealthio_authorization_server::domain::grant::AuthorizationCodeGrant {
                    id: "grant-1".to_string(),
                    client_id: "granted-client".to_string(),
                    scopes: vec!["read".to_string()],
                    granted_at: Utc::now(),
                    last_used_at: None,
                    patient: None,
                    redirect_uri: Url::parse("https://app.example/cb").expect("url"),
                },
            )
        })
        .expect("create grant");
    // Revoke it.
    let res = g
        .router
        .clone()
        .oneshot(build_request(
            Request::delete("/authorization/grants/grant-1")
                .header("host", "127.0.0.1")
                .header("authorization", format!("Bearer {owner_token}")),
            Body::empty(),
        ))
        .await
        .expect("oneshot");
    assert_eq!(res.status(), StatusCode::NO_CONTENT);
    // The client's live access tokens are now revoked via the epoch — a token
    // it holds issued an hour ago reads back revoked.
    assert!(
        revocation_store_handle(&db)
            .is_revoked(
                None,
                Some(Utc::now() - Duration::hours(1)),
                "granted-client"
            )
            .expect("query"),
        "grant revoke must bump the client's revocation epoch"
    );
}
}
