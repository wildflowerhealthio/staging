Wildflower, Read in Order

Session Required

Covers authorization-server-rust/tests/integration/session_required.rs

An app’s request starts in a browser, and the server asks about the browser before it asks about the app. A browser without a browser session gets nothing from /oauth/authorize but a trip to the pairing page, and it comes back to the same request once paired. A browser with one gets the whole flow. Every code its request earns is issued under that browser’s session, never the approver’s, and carries only what that session also holds.

These tests hold the server to both halves. They lean on the browser session, on the code request that records the sid it was made in, and on the launch context a SMART launch carries.

This file’s own helpers are on the Test Fixtures page, with the shared ones.

a_browser_without_a_session_is_sent_to_pairing_even_with_a_standing_grant

A browser without a session is sent to the pairing page, never given a code, though a standing grant covers its request: the FHIR web UI’s client and a hosted app’s alike. Nothing is parked or raised for the Owner, its launch is left unspent, and the pairing page accepts the next that brings it back to the same request.

#![allow(unused)]
fn main() {
async fn a_browser_without_a_session_is_sent_to_pairing_even_with_a_standing_grant() {
    let (g, owner_token, db) = spin_up();
    seed_app(&db);
    for (query, approve_body) in [
        (fhir_ui_query(), APPROVE_FHIR_UI),
        (app_query(), APPROVE_APP),
    ] {
        park_and_approve(
            &g,
            &owner_token,
            &db.paired_browser_token,
            &query,
            approve_body,
        )
        .await;
    }
    let launch = g
        .launch_context_minter
        .mint("test-app")
        .expect("mint a launch");
    let launched_app_query = format!("{}&launch={launch}", app_query());
    let pending_before = owner_prompt(&db);

    let mut nexts = Vec::new();
    for query in [fhir_ui_query(), launched_app_query.clone()] {
        let res = get_authorize_without_session(&g.router, &query).await;
        assert_eq!(res.status(), StatusCode::FOUND, "{query}");
        let next = pairing_next(&location_of(&res));
        assert_eq!(next, format!("/oauth/authorize?{query}"));
        nexts.push(next);
    }
    assert_eq!(owner_prompt(&db), pending_before, "nothing parked");

    for next in nexts {
        let started = start_pairing(&g, &next).await;
        assert_eq!(started.status(), StatusCode::OK);
        assert_eq!(body_json(started.into_body()).await["next"], next);
    }

    // The launch was left for the paired browser, which now takes the fast
    // path with it.
    let res = get_authorize(&g.router, &db.paired_browser_token, &launched_app_query).await;
    assert_eq!(res.status(), StatusCode::FOUND);
    assert!(
        location_of(&res).starts_with("https://app.example/cb?code="),
        "{}",
        location_of(&res)
    );
}
}

a_paired_browser_takes_the_fast_path_under_its_own_session_until_it_is_revoked

A paired browser holding less than the app asks for stops at the limited page, then, going on, takes the fast path on a standing grant another browser earned, and the code, tokens and refresh family name its own session and carry only what that session holds of the grant. Revoking that session ends them, and the app can’t start again from a browser without a session: the revoked browser is sent to pairing.

#![allow(unused)]
fn main() {
async fn a_paired_browser_takes_the_fast_path_under_its_own_session_until_it_is_revoked() {
    let (g, owner_token, db) = spin_up();
    seed_app(&db);
    park_and_approve(
        &g,
        &owner_token,
        &db.paired_browser_token,
        &app_query(),
        APPROVE_APP,
    )
    .await;
    let (sid, session_token) =
        paired_browser_session_holding(&db, &["wildflower/launch", "patient/Patient.rs"]);

    let request_id =
        limited_request_id(&get_authorize(&g.router, &session_token, &app_query()).await);
    let callback = continue_past_session_shortfall(&g.router, &session_token, &request_id).await;
    let issued = redeem(&g, &code_of(&callback)).await;
    assert_eq!(issued["scope"], "patient/Patient.rs offline_access");
    let access_token = issued["access_token"].as_str().expect("an access token");
    let refresh_token = issued["refresh_token"].as_str().expect("a refresh token");
    assert_eq!(sid_of(access_token), sid);
    assert_eq!(
        family_sid(&db, refresh_token).as_deref(),
        Some(sid.as_str())
    );
    let res = post_token(
        &g,
        format!("grant_type=refresh_token&client_id=test-app&refresh_token={refresh_token}"),
    )
    .await;
    assert_eq!(res.status(), StatusCode::OK);
    let refreshed = body_json(res.into_body()).await;
    assert_eq!(
        refreshed["scope"], "patient/Patient.rs offline_access",
        "the refresh family keeps the code's bounded scopes"
    );
    let refresh_token = refreshed["refresh_token"]
        .as_str()
        .expect("a rotated refresh token");

    assert!(store_handle(&db)
        .write(|tx| tx.revoke_browser_session(&sid, Utc::now()))
        .expect("revoke"));
    let res = post_token(
        &g,
        format!("grant_type=refresh_token&client_id=test-app&refresh_token={refresh_token}"),
    )
    .await;
    assert_eq!(res.status(), StatusCode::BAD_REQUEST);
    assert_eq!(body_json(res.into_body()).await["error"], "invalid_grant");
    for res in [
        get_authorize(&g.router, &session_token, &app_query()).await,
        get_authorize_without_session(&g.router, &app_query()).await,
    ] {
        assert_eq!(res.status(), StatusCode::FOUND);
        pairing_next(&location_of(&res));
    }
}
}

the_code_names_the_requesting_browser_s_session_not_the_approver_s

A request a paired browser parks and the Owner approves from another browser (the popup, a signed-in launcher) earns a code, tokens and a refresh family that name the requesting browser’s session, not the approver’s.

#![allow(unused)]
fn main() {
async fn the_code_names_the_requesting_browser_s_session_not_the_approver_s() {

    seed_app(&db);
    let same_device_session_token = signed_in_same_device_session_token(&g).await;
    let request_id = park_and_approve(
        &g,
        &same_device_session_token,
        &db.paired_browser_token,
        &app_query(),
        APPROVE_APP,
    )
    .await;

    let res = get_authorization_status(&g.router, &db.paired_browser_token, &request_id).await;
    assert_eq!(res.status(), StatusCode::OK);

    assert_eq!(status["status"], "approved");

an_owner_approval_from_another_browser_is_bounded_by_the_requesting_session

The Owner approving from another browser grants the request as asked, but the code carries only what the requesting browser’s session also holds: a browser paired without patient/Observation.rs goes on past the limited page and gets a code without it. A request the Owner’s own session makes later takes the fast path on that grant with everything, the same-device session bounding nothing.

#![allow(unused)]
fn main() {
async fn an_owner_approval_from_another_browser_is_bounded_by_the_requesting_session() {
    let (g, _owner_token, db) = spin_up();
    seed_app(&db);
    let same_device_session_token = signed_in_same_device_session_token(&g).await;
    let (_, paired_token) =
        paired_browser_session_holding(&db, &["wildflower/launch", "patient/Patient.rs"]);
    let request_id =
        limited_request_id(&get_authorize(&g.router, &paired_token, &app_query()).await);
    assert_eq!(
        continue_past_session_shortfall(&g.router, &paired_token, &request_id).await,
        format!("/oauth/authorize/{request_id}/wait")
    );
    let res = approve_oauth_consent(
        &g,
        &same_device_session_token,
        &request_id,
        APPROVE_APP.to_owned(),
    )
    .await;
    assert_eq!(res.status(), StatusCode::OK);

    let res = get_authorization_status(&g.router, &paired_token, &request_id).await;
    let status = body_json(res.into_body()).await;
    let issued = redeem(&g, &code_of(status["redirect"].as_str().expect("redirect"))).await;
    assert_eq!(issued["scope"], "patient/Patient.rs offline_access");

    let res = get_authorize(&g.router, &same_device_session_token, &app_query()).await;
    assert_eq!(res.status(), StatusCode::FOUND);
    let issued = redeem(&g, &code_of(&location_of(&res))).await;
    assert_eq!(
        issued["scope"],
        "patient/Patient.rs patient/Observation.rs offline_access"
    );
}
}

a_leaked_request_id_collects_nothing

The status poll answers the browser that parked the request: another paired browser’s session, and no session, get the same 404 as an unknown id, so a leaked request id can’t collect the code.

#![allow(unused)]
fn main() {
async fn a_leaked_request_id_collects_nothing() {
    let (g, owner_token, db) = spin_up();
    seed_app(&db);
    let request_id = park_and_approve(
        &g,
        &owner_token,
        &db.paired_browser_token,
        &app_query(),
        APPROVE_APP,
    )
    .await;
    let (_, other_browser_token) = paired_browser_session(&db);

    let unknown = get_authorization_status(&g.router, &db.paired_browser_token, "no-such-id").await;
    assert_eq!(unknown.status(), StatusCode::NOT_FOUND);
    let unknown_error = body_json(unknown.into_body()).await["error"].clone();
    for res in [
        get_authorization_status(&g.router, &other_browser_token, &request_id).await,
        g.router
            .clone()
            .oneshot(build_request(
                Request::get(format!("/oauth/authorize/{request_id}")),
                Body::empty(),
            ))
            .await
            .expect("oneshot"),
    ] {
        assert_eq!(res.status(), StatusCode::NOT_FOUND);
        assert_eq!(body_json(res.into_body()).await["error"], unknown_error);
    }

    let res = get_authorization_status(&g.router, &db.paired_browser_token, &request_id).await;
    assert_eq!(res.status(), StatusCode::OK);
    assert_eq!(body_json(res.into_body()).await["status"], "approved");
}
}

the_wait_page_reads_the_status_after_the_owner_signs_in_in_the_same_browser

The Owner signs in through /oauth/sign-in in the browser whose wait page is polling (a Custom Tab sharing Chrome’s cookies, a desktop default browser), so the same-device session’s cookie replaces the paired session’s. The poll answers that cookie, whose grant covers wildflower/AuthorizationRequest.r and .u, with the approved redirect, and the code still names the requesting browser’s session. A signed-in session that can read requests but not update them, and no session, get the 404 an unknown id gets.

#![allow(unused)]
fn main() {
async fn the_wait_page_reads_the_status_after_the_owner_signs_in_in_the_same_browser() {
    let (g, _owner_token, db) = spin_up();
    seed_app(&db);
    let request_id =
        parked_request_id(&get_authorize(&g.router, &db.paired_browser_token, &app_query()).await);

    let code = g.same_device_sign_in_code_minter.mint();
    let next = format!("/oauth/authorize/{request_id}/consent");
    let sign_in_uri = format!(
        "/oauth/sign-in?{}",
        url::form_urlencoded::Serializer::new(String::new())
            .append_pair("code", code.as_str())
            .append_pair("next", &next)
            .finish()
    );
    let signed_in = g
        .router
        .clone()
        .oneshot(build_request(
            Request::get(sign_in_uri)
                .header("cookie", browser_session_cookie(&db.paired_browser_token)),
            Body::empty(),
        ))
        .await
        .expect("oneshot");
    let same_device_session_token = browser_session_token(&signed_in);
    assert_ne!(sid_of(&same_device_session_token), db.paired_browser_sid);
    let res = approve_oauth_consent(
        &g,
        &same_device_session_token,
        &request_id,
        APPROVE_APP.to_owned(),
    )
    .await;
    assert_eq!(res.status(), StatusCode::OK);

    let res = get_authorization_status(&g.router, &same_device_session_token, &request_id).await;
    assert_eq!(res.status(), StatusCode::OK);
    let status = body_json(res.into_body()).await;
    assert_eq!(status["status"], "approved");
    let callback = status["redirect"].as_str().expect("the approved redirect");
    assert!(
        callback.starts_with("https://app.example/cb?"),
        "{callback}"
    );
    let issued = redeem(&g, &code_of(callback)).await;
    let access_token = issued["access_token"].as_str().expect("an access token");
    assert_eq!(sid_of(access_token), db.paired_browser_sid);

    let unknown =
        get_authorization_status(&g.router, &same_device_session_token, "no-such-id").await;
    assert_eq!(unknown.status(), StatusCode::NOT_FOUND);
    let unknown_error = body_json(unknown.into_body()).await["error"].clone();
    let cannot_decide = signed_in_session_token_that_cannot_decide(&g, &db).await;
    for res in [
        get_authorization_status(&g.router, &cannot_decide, &request_id).await,
        g.router
            .clone()
            .oneshot(build_request(
                Request::get(format!("/oauth/authorize/{request_id}")),

            .await
            .expect("oneshot"),
    ] {
        assert_eq!(res.status(), StatusCode::NOT_FOUND);
        assert_eq!(body_json(res.into_body()).await["error"], unknown_error);
    }
}

an_expired_launch_and_an_unknown_one_go_back_to_the_app_saying_why

A launch for this client that expired (as one may while the Owner approves the browser’s pairing) goes back to the app as invalid_request, told to open the app again; a launch this server never minted is told it isn’t recognized.

#![allow(unused)]
fn main() {
async fn an_expired_launch_and_an_unknown_one_go_back_to_the_app_saying_why() {
    let (g, _owner_token, db) = spin_up();
    seed_app(&db);
    store_handle(&db)
        .write(|tx| {
            tx.insert_launch_context(&LaunchContext::for_client(
                "lapsed".to_owned(),
                "test-app",
                None,
                Utc::now() - LAUNCH_CONTEXT_TTL,
            ))
        })
        .expect("plant a lapsed launch");

    let res = get_authorize(
        &g.router,
        &db.paired_browser_token,
        &format!("{}&launch=lapsed", app_query()),
    )
    .await;
    assert_eq!(res.status(), StatusCode::FOUND);
    assert_eq!(
        location_of(&res),
        "https://app.example/cb?error=invalid_request\
         &error_description=This+launch+has+expired.+Open+the+app+again.&state=xyz"
    );

    let res = get_authorize(
        &g.router,
        &db.paired_browser_token,
        &format!("{}&launch=never-minted", app_query()),
    )
    .await;
    assert_eq!(res.status(), StatusCode::FOUND);
    assert_eq!(
        location_of(&res),
        "https://app.example/cb?error=invalid_request&error_description=\
         This+launch+isn%27t+recognized.+Open+the+app+again.&state=xyz"
    );
}
