pub fn spin_up_with_authorization_start_gate(
    authorization_start_gate: Arc<dyn AuthorizationStartGate>,
) -> (AuthorizationServer, String, TestDb) {
    // One shared in-memory diesel pool, built once and handed to the slice —
    // mirrors how the host wires the app-wide `wildflowerhealthio_persistence::open_pool`
    // pool into each diesel-backed slice. `db.pool` is that shared handle;
    // `store_handle` clones it to reach the same database.
    let pool = wildflowerhealthio_persistence::open_in_memory_pool().expect("open in-memory pool");
    let config = AuthorizationServerConfig {
        server_origin: Url::parse(SERVER_ORIGIN).expect("SERVER_ORIGIN is a valid URL"),
        host_app_client_scopes:
            wildflowerhealthio_authorization_server::default_host_app_client_scopes(),
        fhir_ui_client_id: "fhir-ui-client".to_string(),
        launcher_base: LauncherBase::parse(LAUNCHER_BASE).expect("LAUNCHER_BASE is a valid URL"),
        authorization_app_base: AuthorizationAppBase::parse(AUTHORIZATION_APP_BASE)
            .expect("AUTHORIZATION_APP_BASE is a valid URL"),
        launcher_url: Url::parse(LAUNCHER_URL).expect("LAUNCHER_URL is a valid URL"),
        authorization_url: Url::parse(AUTHORIZATION_URL).expect("AUTHORIZATION_URL is a valid URL"),
    };
    let (pending_consent_tx, pending_consent_rx) = watch::channel::<Option<OwnerPrompt>>(None);
    // The shared revocation store lives on the rusqlite connection the host
    // wires into both the authorization server and the FHIR server (#269). A second handle on
    // the same connection (see `revocation_store_handle`) lets tests
    // plant/observe rows.
    let revocation_conn = Connection::open_in_memory().expect("open revocation db");
    let revocation_store =
        wildflowerhealthio_token_revocation::RevocationStore::new(revocation_conn.clone())
            .expect("revocation store");
    let g = setup_authorization_server(
        pool.clone(),
        revocation_store,
        &config,
        pending_consent_tx,
        authorization_start_gate,
    )
    .expect("setup");
    let mut db = TestDb {
        pool,
        revocation_conn,
        pending_consent_rx,
        paired_browser_token: String::new(),
        paired_browser_sid: String::new(),
    };
    (db.paired_browser_sid, db.paired_browser_token) = paired_browser_session(&db);
    let owner_token = mint_owner_token(&db);
    (g, owner_token, db)
}

plant_authorization_code

Plant an Approved code-flow request plus its redeemable AuthorizationCode straight in the store, made in the browser session sid. expires_at (shared by both rows) lets a caller force an already-expired code without sleeping. The challenge is the real S256 digest of CODE_VERIFIER.

#![allow(unused)]
fn main() {
pub fn plant_authorization_code(
    store: &SqliteGrantStore,
    sid: &str,
    client_id: &str,
    redirect_uri: &Url,
    scopes: &[&str],
    code: &str,
    expires_at: chrono::DateTime<Utc>,
) {
    let request_id = format!("req-{code}");
    let scope_vec: Vec<String> = scopes.iter().map(ToString::to_string).collect();
    let challenge = compute_code_challenge(CODE_VERIFIER);
    let now = Utc::now();
    let requested_at = expires_at.min(now) - Duration::seconds(1);
    let request = CodeRequest::new(
        NewCodeRequestArgs {
            id: request_id.clone(),
            client_id: client_id.to_string(),
            requested_scopes: scope_vec.clone(),
            ttl: expires_at - requested_at,
            redirect_uri: redirect_uri.clone(),
            code_challenge: challenge.clone(),
            client_state: "state".to_string(),
            sid: sid.to_owned(),
            launch: None,
            launch_bound_patient: None,
            pre_approved_scopes: Vec::new(),
            decidable_in_requesting_browser: false,
            awaits_limited_page: false,
        },
        requested_at,
    )
    .approve(
        Approval {
            granted_scopes: scope_vec.clone(),
            patient: None,
        },
        requested_at,
    )
    .expect("a pending request approves");
    store
        .write(|tx| tx.insert_code_request(&request))
        .expect("insert request");
    store
        .write(|tx| {
            tx.issue_authorization_code(&AuthorizationCode {
                code: code.to_string(),
                request_id,
                client_id: client_id.to_string(),
                redirect_uri: redirect_uri.clone(),
                code_challenge: challenge,
                granted_scopes: scope_vec,
                patient: None,
                issued_at: now,
                expires_at,
                sid: sid.to_owned(),
            })
        })
        .expect("issue code");
}
}

plant_device_request

Plant a device-flow request directly, keyed by WILD-FLWR. status and expires_at are caller-controlled so the device state-machine tests can stand up approved or expired rows the public surface can’t mint on demand: the request starts pending just before it would expire (or now) and moves to status, an approval granting the requested scopes.

#![allow(unused)]
fn main() {
pub fn plant_device_request(
    store: &SqliteGrantStore,
    client_id: &str,
    device_code: &str,
    scopes: &[&str],
    status: RequestStatus,
    expires_at: chrono::DateTime<Utc>,
) {
    let scope_vec: Vec<String> = scopes.iter().map(ToString::to_string).collect();
    let requested_at = expires_at.min(Utc::now()) - Duration::seconds(1);
    let pending = DeviceRequest::new(
        NewDeviceRequestArgs {
            id: device_code.to_string(),
            client_id: client_id.to_string(),
            requested_scopes: scope_vec.clone(),
            ttl: expires_at - requested_at,
            user_code: "WILD-FLWR".to_string(),
            device_name: None,
        },
        requested_at,
    );
    let approve = |request: &DeviceRequest| {
        request
            .approve(
                Approval {
                    granted_scopes: scope_vec.clone(),
                    patient: None,
                },
                None,
                requested_at,
            )
            .expect("a pending request approves")
    };
    let request = match status {
        RequestStatus::Pending => pending,
        RequestStatus::Approved => approve(&pending),
        RequestStatus::Denied => pending
            .deny(requested_at)
            .expect("a pending request denies"),
        RequestStatus::Claimed => approve(&pending)
            .claim(requested_at)
            .expect("an approved request is claimed"),
        RequestStatus::AwaitingLimitedPage => panic!("a device request has no limited page"),
    };
    store
        .write(|tx| tx.insert_device_request(&request))
        .expect("insert device request");
}
}
