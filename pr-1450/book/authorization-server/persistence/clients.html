Clients

Covers authorization-server-rust/src/db/clients.rs

The clients table holds every app’s registration: one row per client_id, read back as a Client, which maps onto the row one field to one column of the same name. The row is the registration. Every check at /oauth/authorize and /oauth/token reads it, the Owner’s approvals of an app trusted on first use write it, and boot-time seeding rewrites the server’s own clients on every boot. The primary key is client_id, so an app has at most one registration, and every lookup goes through that key.

The three list columns hold compact JSON text: allowed_scopes through the shared JsonStrings, and redirect_uris and allowed_grant_types through two mappings only the client binds, JsonUrls and JsonAllowedGrantTypes. kind and registration_source are stored as their wire strings. The tests draw clients from arb_client.

Each of the client row’s custom column mappings rejects an out-of-domain stored value on read as a typed diesel error, never a panic: the JSON TEXT newtypes (JsonUrls / JsonStrings / JsonAllowedGrantTypes) on malformed JSON or an unknown enum member, and the ClientKind text-enum mapping on an unknown discriminant. A row tampered via raw SQL surfaces at the client_by_id read boundary as Err, so a corrupt row can never silently decode to a wrong-but-valid client.

#![allow(unused)]
fn main() {
fn stored_columns_reject_corrupt_values() {
    // `column = bad_value` tampered onto the migration-seeded `growth_chart`
    // row, each asserted to fail the read.
    for (column, bad_value) in [
        ("redirect_uris", "not json"),        // JsonUrls: malformed JSON
        ("redirect_uris", "[\"not a url\"]"), // JsonUrls: valid JSON, not a URL
        // JsonUrls: a path entry is not an absolute URL (why migration 0021
        // strips them)
        ("redirect_uris", "[\"/\"]"),
        ("allowed_scopes", "not json"), // JsonStrings: malformed JSON
        ("allowed_grant_types", "[\"totally_unknown\"]"), // JsonAllowedGrantTypes: unknown member
        ("kind", "bogus_kind"),                           // ClientKind: unknown discriminant
    ] {
        let store = SqliteGrantStore::open_in_memory().expect("open in-memory store");
        let mut conn = store.pool().get().expect("check out a connection");
        diesel::sql_query(format!(
            "UPDATE clients SET {column} = '{bad_value}' WHERE client_id = 'growth_chart'"
        ))
        .execute(&mut conn)
        .expect("tamper the stored row");
        drop(conn);
        assert!(
            store.read(|tx| tx.client_by_id("growth_chart")).is_err(),
            "a corrupt `{column}` must surface as a typed read error",
        );
    }
}
}

upsert_client

An app keeps the time it was first registered, and a disabled app stays disabled, whoever writes its registration again: an approval that widens a disabled app’s registration doesn’t re-enable it.

Insert a client, or update its policy fields if one with the same client_id already exists. Used by the seeding every boot runs, so a seeded client’s definition always matches the code, and by the grant recorder’s trust-on-first-use registration. Uses ON CONFLICT … DO UPDATE, so it never deletes the row and preserves registered_at and disabled_at — the original registration time and any admin disable survive rather than the client being resurrected.

#![allow(unused)]
fn main() {
fn upsert_client(&mut self, client: &Client) -> Result<(), AuthorizationServerError> {
    let conn = &mut *self.conn;
    use diesel::upsert::excluded;
    diesel::insert_into(clients::table)
        .values(client.clone())
        .on_conflict(clients::client_id)
        .do_update()
        .set((
            clients::name.eq(excluded(clients::name)),
            clients::kind.eq(excluded(clients::kind)),
            clients::redirect_uris.eq(excluded(clients::redirect_uris)),
            clients::allowed_scopes.eq(excluded(clients::allowed_scopes)),
            clients::allowed_grant_types.eq(excluded(clients::allowed_grant_types)),
            clients::secret_hash.eq(excluded(clients::secret_hash)),
            clients::registration_source.eq(excluded(clients::registration_source)),
        ))
        .execute(conn)
        .map_err(|e| AuthorizationServerError::infrastructure("upsert_client failed", e))?;
    Ok(())
}
}

fn main() {
fn upsert_updates_policy_but_preserves_registration_and_disable() {
    let store = SqliteGrantStore::open_in_memory().expect("open in-memory store");
    let registered_at = chrono::DateTime::from_timestamp(1_000, 0).unwrap();
    let disabled_at = chrono::DateTime::from_timestamp(1_500, 0).unwrap();
    let mut client = Client {
        client_id: "c1".to_string(),
        name: "First".to_string(),
        kind: ClientKind::Public,
        redirect_uris: vec![],
        allowed_scopes: vec!["openid".to_string()],
        allowed_grant_types: AllowedGrantType::ALL.to_vec(),
        secret_hash: None,
        registered_at,
        disabled_at: Some(disabled_at),
        registration_source: RegistrationSource::TrustedOnFirstUse,
    };
    store.write(|tx| tx.upsert_client(&client)).expect("insert");

    // Re-seed with drifted policy, a newer registered_at, and disabled_at
    // cleared: the policy fields update, but registration time and the admin
    // disable are preserved (the row is updated in place, never resurrected).
    client.name = "Renamed".to_string();
    client.allowed_scopes = vec!["system/*.cruds".to_string()];
    client.registered_at = chrono::DateTime::from_timestamp(2_000, 0).unwrap();
    client.disabled_at = None;
    client.registration_source = RegistrationSource::ServerDefined;
    store.write(|tx| tx.upsert_client(&client)).expect("update");

    let fetched = store
        .read(|tx| tx.client_by_id("c1"))
        .expect("query")
        .expect("row present");
    assert_eq!(fetched.name, "Renamed");
    assert_eq!(fetched.allowed_scopes, vec!["system/*.cruds".to_string()]);
    assert_eq!(fetched.registered_at, registered_at);
    assert_eq!(fetched.disabled_at, Some(disabled_at));
    assert_eq!(
        fetched.registration_source,
        RegistrationSource::ServerDefined,
        "seeding marks the rows it owns server-defined"
    );
}
}

#![allow(unused)]
fn main() {
fn list_clients_returns_every_row_sorted_by_id(
    upserted in prop::collection::vec(arb_client(), 0..6),
) {
    let store = SqliteGrantStore::open_in_memory().expect("open in-memory store");
    let mut expected: std::collections::BTreeMap<String, Client> = store
        .read(|tx| tx.list_clients())
        .expect("list seeds")
        .into_iter()
        .map(|c| (c.client_id.clone(), c))
        .collect();
    for client in &upserted {
        // Fresh ids only: a repeat upsert keeps the first row's
        // `registered_at` / `disabled_at` by design, which is
        // `upsert_client`'s property, not this one's.
        if !expected.contains_key(&client.client_id) {
            store.write(|tx| tx.upsert_client(client)).expect("upsert");
            expected.insert(client.client_id.clone(), client.clone());
        }
    }

    let listed = store.read(|tx| tx.list_clients()).expect("list");
    prop_assert_eq!(listed, expected.into_values().collect::<Vec<_>>());
}
}

#![allow(unused)]
fn main() {
fn set_client_disabled_round_trips_only_the_stamp(
    client in arb_client(),
    stamp in arb_timestamp(),
) {
    let store = SqliteGrantStore::open_in_memory().expect("open in-memory store");
    // A seeded id would keep the seed's policy on upsert; use a fresh one.
    prop_assume!(!seeded_client_ids(&store).contains(&client.client_id));
    store.write(|tx| tx.upsert_client(&client)).expect("upsert");

    prop_assert!(store.write(|tx| tx.set_client_disabled(&client.client_id, Some(stamp))).expect("disable"));
    let disabled = store.read(|tx| tx.client_by_id(&client.client_id)).expect("query").expect("row");
    prop_assert_eq!(&disabled, &Client { disabled_at: Some(stamp), ..client.clone() });

    prop_assert!(store.write(|tx| tx.set_client_disabled(&client.client_id, None)).expect("enable"));
    let enabled = store.read(|tx| tx.client_by_id(&client.client_id)).expect("query").expect("row");
    prop_assert_eq!(enabled, Client { disabled_at: None, ..client });
}
}

The seeded clients

The SMART apps’ clients aren’t written by Rust: SQL migrations insert their rows. These tests are the guard that the hand-written SQL decodes as a Client. Several also pin a row to the app it registers, read from that app’s own source, so the row and the scopes the app asks for can’t drift apart unnoticed.

Every SMART client is seeded by a migration (not Rust) — the sample apps by 0003, Medications by 0004 (renamed and given its published-site redirect by 0006), the server-docs API console by 0007, the Importer by 0008, the OHIF imaging viewer by 0009 (re-seeded by 0016 on an install that skipped it), FHIR Sync for Pebble by 0017 (its Observation scope widened by 0018), Strength by 0019, Plots by 0022 (both renamed by 0035), and the Synthetic Data Loader by 0028 (which re-keys Medications, the console, the Importer and Plots to random ids, as 0025 / 0026 did Strength and the launcher, 0029 does the OHIF viewer and 0030 FHIR Sync for Pebble), and Wildflower Authorization by 0031 (the launcher’s scopes are narrowed by 0032), and 0037 puts each hosted app’s own origin in front of the redirect on wildflowerhealth.io it held — so a freshly-migrated store has them all, and every hand-written row decodes back to a valid Client. This is the guard that the SQL seeds’ JSON columns and registered_at text stay in the exact shape the store’s read path parses (a malformed value would fail client_by_id’s row mapping, not silently).

The marketing site publishes each first-party app’s public OAuth client registration in suggested_apps.json, for a server that registers the app itself. #![allow(unused)]
fn main() {
fn suggested_apps_json_publishes_the_seeded_first_party_clients() {
    const SUGGESTED_APPS_JSON: &str = include_str!(concat!(
        env!("CARGO_MANIFEST_DIR"),
        "/../../../../apps/wildflower-site/marketing-site-web/public/suggested_apps.json"
    ));
    let document: serde_json::Value =
        serde_json::from_str(SUGGESTED_APPS_JSON).expect("suggested_apps.json is JSON");
    let store = SqliteGrantStore::open_in_memory().expect("open in-memory store");
    let mut checked = 0;
    for app in document["apps"].as_array().expect("an apps array") {
        let Some(published) = app.get("oauthClient") else {
            continue;
        };
        if app["publisher"]["kind"] != "first-party" {
            continue;
        }
        let id = app["id"].as_str().expect("an id");
        let client_id = published["clientId"].as_str().expect("a clientId");
        let client = store
            .read(|tx| tx.client_by_id(client_id))
            .expect("query (a decode failure surfaces here)")
            .unwrap_or_else(|| panic!("{id}: {client_id} is seeded by a clients migration"));

        assert_eq!(client.kind, ClientKind::Public, "{id}");
        assert!(client.secret_hash.is_none(), "{id}");
        assert_eq!(published["tokenEndpointAuthMethod"], "none", "{id}");
        let scopes: Vec<String> =
            serde_json::from_value(app["scopes"].clone()).expect("scopes are strings");
        assert_eq!(scopes, client.allowed_scopes, "{id}: scopes");
        let grant_types: Vec<AllowedGrantType> =
            serde_json::from_value(published["grantTypes"].clone()).expect("grant types");
        assert_eq!(grant_types, client.allowed_grant_types, "{id}: grantTypes");
        for redirect in published["redirectUris"].as_array().expect("redirectUris") {
            let redirect = url::Url::parse(redirect.as_str().expect("a string"))
                .expect("an absolute redirect");
            assert!(
                client.redirect_uris.contains(&redirect),
                "{id}: {redirect} is not registered for {client_id}",
            );
        }
        checked += 1;
    }
    assert!(
        checked > 0,
        "suggested_apps.json lists no first-party OAuth client"
    );
}
